const Part2 = () => {
  const frame = useCurrentFrame();
  const showParts = Math.min(3, Math.floor(frame / 15));
  return (
    <Stage>
      <Title>逐步涂色</Title>
      <div style={{
        position: "absolute",
        left: 500,
        top: 300,
        width: 600,
        height: 120,
        border: "3px solid #3b3430",
        display: "grid",
        gridTemplateColumns: "repeat(4, 1fr)"
      }}>
        {[1,2,3,4].map(i => <div key={i} style={{
          borderRight: i<4 ? "2px dashed #6d645a" : "none",
          background: i <= showParts ? "#b0501e" : "transparent",
          transition: "background 0.3s"
        }}/>)}
      </div>
      <Body x={120} y={300} w={800}>
        {showParts >= 1 && "已涂"+showParts+"份 → "}
        {showParts >= 1 && <Em>四分之{showParts}</Em>}
      </Body>
    </Stage>
  );
};
